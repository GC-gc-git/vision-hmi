using System;

using System.IO;

namespace w1_day2_日志级别及统一格式
{
    /// <summary>
    /// 应用日志器
    /// </summary>
    public static class AppLogger
    {
        public enum LogLevel { Debug = 0, Info = 1, Error = 2 }
        public static LogLevel MinLevel = LogLevel.Debug;
        /// <summary>
        /// 项目细节
        /// </summary>
        /// <param name="msg"></param>
        public static void Logdebug(string msg)
        {
            try
            {
                string dir = Path.Combine(AppDomain.CurrentDomain.BaseDirectory, "logs");
                if (Directory.Exists(dir))
                {
                    Directory.CreateDirectory(dir);
                }
                string file = Path.Combine(dir, $"debug{DateTime.Now.ToString("yyyy-MM-dd")}");
                File.AppendAllText(file, $"[{DateTime.Now.ToString("yyyy-MM-dd HH-mm-ss")}][DEBUG][{msg}被点击]");
            }
            catch (Exception ex2)
            {
                System.Diagnostics.Debug.WriteLine(ex2);
            }
        }
        /// <summary>
        /// 项目运行节点(整个项目干了什么)
        /// </summary>
        /// <param name="msg"></param>
        public static void LogInfo(string msg)
        {
            try
            {
                string dir = Path.Combine(AppDomain.CurrentDomain.BaseDirectory, "logs");
                if (Directory.Exists(dir))
                {
                    Directory.CreateDirectory(dir);
                }
                string file = Path.Combine(dir, $"Info{DateTime.Now.ToString("yyyy-MM-dd")}");
                File.AppendAllText(file, $"[{DateTime.Now.ToString("yyyy-MM-dd HH-mm-ss")}][INFO][{msg}被打开]");
            }
            catch (Exception ex2)
            {
                System.Diagnostics.Debug.WriteLine(ex2);
            }
        }
        /// <summary>
        /// 异常报错存放处
        /// </summary>
        /// <param name="whereText"></param>
        /// <param name="ex"></param>
        public static void LogError(string whereText,Exception ex)
        {
            try
            {
                string dir = Path.Combine(AppDomain.CurrentDomain.BaseDirectory, "logs");
                if (Directory.Exists(dir))
                {
                    Directory.CreateDirectory(dir);
                }
                string path = Path.Combine(dir, $"error{DateTime.Now.ToString("yyyy-MM-dd")}.txt");
                File.AppendAllText(path, $"[{DateTime.Now.ToString("yyyy-MM-dd HH:mm:ss.fff")}][ERROR][{whereText}]{ex}\n", Encoding.UTF8);
            }
            catch(Exception ex2)
            {
                System.Diagnostics.Debug.WriteLine(ex2);
            }
        }
    }
}

【已通过评审修正的点】

- 缺整体 try-catch -> 日志失败不能盖掉原始异常

- {msg}被打开 / {msg}被点击 -> 日志类不该知道 UI 语义，改成 [{msg}]，语义由调用方给
